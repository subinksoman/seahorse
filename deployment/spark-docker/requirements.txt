# Executor image Python deps — Python 3.12 (Miniconda py312) + Spark 3.4.4 (pyspark from $SPARK_HOME).
# The Python-3.7-era pins below were unpinned/bumped so wheels exist for cp312; C-extension
# libraries are pinned to their first Python-3.12-compatible line.

# Jupyter / IPython kernel stack (the notebook pyspark kernel connects to this image)
ipykernel
ipython
jupyter-client
jupyter-core
pyzmq
tornado
traitlets

# Core numerics — keep numpy < 2: Spark 3.4's Arrow<->pandas bridge and the ML libs below
# are not validated against the numpy 2.0 ABI break.
numpy<2
# Spark 4.2's PySpark requires pandas >= 2.2.0 (require_minimum_pandas_version). Spark 3.4.4 also
# works with 2.2.x (it only enforces a lower bound). Cap below the pandas 3.0 rewrite.
pandas>=2.2,<3
python-dateutil
pytz
requests
urllib3
wcwidth

# ML / analytics
scikit-learn
scipy
seaborn
shap
reportlab
# SQLAlchemy 1.4.45 has no cp312 wheel; 1.4.54 is the last 1.4 line and keeps PyHive 0.7 working.
SQLAlchemy==1.4.54
xgboost
tensorflow
keras
kmodes==0.12.2
flask
matplotlib

# Arrow: pyspark <-> pandas conversion. Spark 4.2 requires PyArrow >= 18.0.0 for the optimized
# Arrow path (with 15.x it warns and falls back to slow non-Arrow). 18.x has cp312 wheels.
pyarrow>=18.0.0

# DB / connectors
oracledb==2.3.0
PyHive==0.7.0
# lightgbm 3.3.5 has no cp312 wheel and its source build relies on distutils (removed in 3.12).
# 4.x ships cp312 wheels and reads 3.x models; train/predict API is unchanged.
lightgbm==4.3.0
